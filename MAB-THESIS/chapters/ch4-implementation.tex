\chapter{پیاده‌سازی و ارزیابی نتایج}
\label{ch:implementation}

در این فصل، جزئیات محاسباتی، ساختار الگوریتمی و نتایج تجربی حاصل از پیاده‌سازی و آموزش چارچوب شبکه‌های عصبی آگاه از فیزیک در حل مستقیم معادله غیرخطی برگرز ارائه می‌گردد. ابتدا بستر نرم‌افزاری و ابزارهای توسعه معرفی شده و سپس ساختار الگوریتمی و روند گام‌به‌گام آموزش مدل با بهره‌گیری از قابلیت‌های کتابخانه جکس تشریح می‌شود. در ادامه، جزئیات پیکربندی مدل زمان‌پیوسته، تاریخچه همگرایی بهینه‌سازی ترکیبی دو مرحله‌ای، توزیع مکانی-زمانی خطاها و نحوه به دام انداختن جبهه تیز شوک گزارش و تحلیل می‌گردند. در بخش پایانی فصل، ارزیابی تطبیقی جامعی میان یافته‌های این پژوهش و نتایج مقاله معتبر مرجع رئیسی و همکاران (۲۰۱۹) به همراه تحلیل نقاط قوت و محدودیت‌های روش ارائه می‌شود.

\section{بستر محاسباتی و ابزارهای پیاده‌سازی}
\label{sec:env}

کلیه الگوریتم‌های محاسباتی این پژوهش با زبان برنامه‌نویسی پایتون نسخه ۳.۱۱ پیاده‌سازی شده‌اند. برای پیاده‌سازی ساختار شبکه عصبی، کامپایل توابع و انجام مشتق‌گیری خودکار، کتابخانه جکس\LTRfootnote{JAX} به کار گرفته شده است \cite{jax2018}. جکس امکان ترکیب تبدیل‌های تابعی نظیر مشتق‌گیری خودکار (\lr{grad})، برداری‌سازی خودکار (\lr{vmap}) و کامپایل در زمان اجرا (\lr{jit}) را بر بستر کامپایلر \lr{XLA} فراهم می‌سازد. انجام محاسبات عددی و ماتریسی با استفاده از کتابخانه‌های نامپای و سای‌پای، و رسم نمودارها با کتابخانه مت‌پلات‌لیب صورت پذیرفته است. الگوریتم بهینه‌سازی ترکیبی شامل پیاده‌سازی مستقیم بهینه‌ساز آدام و استفاده از رابط الگوریتم بهینه‌سازی مقید \lr{L-BFGS-B} در کتابخانه سای‌پای است.

تمامی مراحل آموزش و آزمایش‌های عددی با دقت ممیز شناور ۶۴ بیتی (\lr{Float64}) اجرا شده‌اند تا خطاهای محاسباتی ناشی از گردکردن اعداد به کمترین حد ممکن برسد و امکان بازتولید دقیق نتایج بر روی تجهیزات استاندارد دانشگاهی فراهم باشد. به‌منظور تضمین تکرارپذیری، بذرهای تولید اعداد تصادفی ثابت نگاه داشته شده و کد کامل برنامه به همراه راهنمای اجرا در پیوست الف و پوشه اختصاصی \lr{code} ارائه گردیده است.

\section{ساختار الگوریتمی و روند آموزش مدل با کتابخانه جکس}
\label{sec:jax-algo}

فرآیند آموزش شبکه عصبی آگاه از فیزیک برای حل مستقیم معادله غیرخطی برگرز شامل یک جریان داده پیوسته و چندمرحله‌ای است که از تولید نقاط آموزشی آغاز شده و تا همگرایی نهایی پارامترها ادامه می‌یابد. در ادامه، ساختار الگوریتمی این فرآیند به‌تفصیل تبیین می‌گردد:

\subsection{مقداردهی اولیه پارامترها با روش گزاویه}
بردار پارامترهای مدل $\theta = \{\mathbf{W}^{(l)}, \mathbf{b}^{(l)}\}_{l=1}^L$ شامل ماتریس‌های وزن و بردارهای بایاس است. برای جلوگیری از پدیده انفجار یا محوشدگی گرادیان در آغاز آموزش، ماتریس وزن لایه $l$ام با ابعاد $n_{\text{in}} \times n_{\text{out}}$ با توزیع یکنواخت گزاویه (گلوروت) مقداردهی می‌شود \cite{glorot2010}:
\begin{equation}
\mathbf{W}^{(l)} \sim \mathcal{U}\left[-\sqrt{\frac{6}{n_{\text{in}} + n_{\text{out}}}},\, \sqrt{\frac{6}{n_{\text{in}} + n_{\text{out}}}}\right]
\end{equation}
و کلیه بایاس‌ها $\mathbf{b}^{(l)}$ با صفر مقداردهی اولیه می‌گردند.

\subsection{خط لوله تبدیلات تابعی در JAX}
در کتابخانه جکس، تمام فرآیندها به‌صورت توابع محض پیاده‌سازی می‌شوند:
\begin{enumerate}
\item \textbf{تابع پیش‌بینی حالت ($u_\theta$):} تابع \lr{u\_net(params, t, x)} ورودی‌های اسکالر $(t,x)$ را دریافت کرده و با گذر از لایه‌های پنهان با تابع فعال‌ساز $\tanh$، مقدار اسکالر سرعت را برمی‌گرداند.
\item \textbf{استخراج مشتقات جزئی با \lr{jax.grad}:} مشتقات زمانی و مکانی با اعمال تبدیل تابعی استخراج می‌گردند:
\begin{equation}
\begin{aligned}
u_t &= \text{\lr{jax.grad(u\_net, argnums=1)(params, t, x)}} \\
u_x &= \text{\lr{jax.grad(u\_net, argnums=2)(params, t, x)}} \\
u_{xx} &= \text{\lr{jax.grad(jax.grad(u\_net, argnums=2), argnums=2)(params, t, x)}}
\end{aligned}
\end{equation}
\item \textbf{بردارسازی با \lr{jax.vmap}:} توابع فوق که برای یک نقطه اسکالر تعریف شده‌اند، با استفاده از تبدیل \lr{vmap} به توابع برداری تبدیل می‌شوند که قادرند هم‌زمان روی دسته‌های بزرگ نقاط هم‌مکانی بدون نیاز به حلقه‌های پایتون اجرا گردند.
\item \textbf{کامپایل در زمان اجرا با \lr{jax.jit}:} توابع هزینه و گرادیان با کامپایلر \lr{XLA} به کدهای ماشین بهینه‌سازی‌شده تبدیل می‌شوند که سرعت اجرای آموزش را تا بیش از ۱۰ برابر افزایش می‌دهد.
\end{enumerate}

\subsection{استراتژی بهینه‌سازی ترکیبی دو مرحله‌ای}
رویه آموزش به منظور عبور از کمینه‌های محلی و دستیابی به بیشترین دقت در دو فاز متوالی اجرا می‌شود:
\begin{itemize}
\item \textbf{فاز اول (بهینه‌ساز آدام):} اجرای ۶۰۰۰ گام بهینه‌سازی با نرخ یادگیری $\eta = 10^{-3}$ و پارامترهای گشتاور $\beta_1=0.9$ و $\beta_2=0.999$. در این فاز، خاصیت تطبیقی آدام موجب فرار سریع مدل از نواحی نامطلوب و کاهش خطای کلی به مرتبه $10^{-3}$ می‌گردد.
\item \textbf{فاز دوم (بهینه‌ساز شبه‌نیوتنی \lr{L-BFGS}):} با استفاده از خروجی فاز اول به‌عنوان نقطه شروع، بهینه‌سازی با الگوریتم \lr{L-BFGS-B} با جستجوی خطی ولف قوی ادامه می‌یابد. این الگوریتم با تقریب مرتبه دو ماتریس معکوس هشین، همگرایی نهایی به دقت‌های بسیار بالا را رقم می‌زند.
\end{itemize}

\section{حل مستقیم معادله برگرز}
\label{sec:forward}

در مسئله مستقیم، هدف محاسبه جواب $u(t,x)$ در دامنه مکانی $x \in [-1,1]$ و زمانی $t \in [0,1]$ با شرط اولیه $u(0,x) = -\sin(\pi x)$ و شرایط مرزی همگن دیریکله $u(t,-1) = u(t,1) = 0$ و ضریب لزجت $\nu = 0.01/\pi$ است. ساختار شبکه عصبی مورد استفاده، پرسپترون چندلایه با ۴ لایه پنهان، ۵۰ نورون در هر لایه و تابع فعال‌ساز $\tanh$ است که در مجموع شامل ۷۸۵۱ پارامتر قابل آموزش می‌باشد.

داده‌های آموزشی روی مرزها و شرط اولیه شامل ۲۰۰ نقطه ($N_u = 200$) شامل ۱۰۰ نقطه در $t=0$ و ۱۰۰ نقطه روی مرزهای مکانی است. تعداد نقاط هم‌مکانی درون دامنه برابر ۱۰۰۰۰ نقطه ($N_f = 10000$) انتخاب شد که نیمی از آن‌ها به‌صورت تصادفی یکنواخت و نیمی دیگر با توزیع متمرکز حول خط $x=0$ (ناحیه تشکیل جبهه شوک) نمونه‌برداری گردیدند. موقعیت نقاط آموزشی در شکل \ref{fig:fwdpoints} و مشخصات ساختاری آموزش در جدول \ref{tab:fwdconfig} خلاصه شده است.

بررسی نتایج مقایسه‌ای نکات مهم زیر را نشان می‌دهد:
\begin{enumerate}
\item \textbf{عمق در برابر پهنای شبکه:} در مقاله مرجع از شبکه‌ای عمیق‌تر (۹ لایه پنهان) با پهنای کم (۲۰ نورون) استفاده شده است، در حالی که در پژوهش حاضر از معماری با ۴ لایه پنهان و ۵۰ نورون استفاده گردید. معماری‌های عمیق‌تر قابلیت بالاتری در تقریب نواحی با گرادیان‌های بسیار تیز و ناپیوستگی‌های موضعی دارند که دلیل اصلی تفاوت در مرتبه خطای نهایی است.
\item \textbf{سقف تکرارهای بهینه‌سازی و سخت‌افزار:} در مقاله مرجع بهینه‌سازی \lr{L-BFGS} با تعداد تکرارهای بسیار بالا بر روی پردازنده‌های گرافیکی تخصصی تا رسیدن به حد نهایی دقت اجرا شده است، در حالی که در پژوهش حاضر به‌منظور امکان اجرای روان بر روی پردازنده‌های مرکزی معمولی دانشگاهی، تعداد تکرارها کنترل شده است.
\item \textbf{کیفیت بازسازی پروفیل شوک:} با وجود تفاوت عددی در خطای میانگین، همان‌طور که در شکل \ref{fig:snapshots} مشاهده می‌شود، مدل پیاده‌سازی‌شده در این پژوهش موقعیت زمانی و مکانی جبهه شوک و شیب شدید آن را در مقاطع مختلف $t=0.25, 0.50, 0.75$ با وفاداری کیفی و کمی بسیار بالایی بازتولید نموده است.
\end{enumerate}

\section{بحث و تحلیل نتایج}
\label{sec:discussion}

تحلیل نتایج عددی حاصل از پیاده‌سازی، ابعاد علمی و مهندسی متعددی را در خصوص رفتار شبکه‌های آگاه از فیزیک در حل معادله برگرز روشن می‌سازد:

\textbf{کارایی داده‌ای فوق‌العاده.} در مسئله مستقیم، تقریب میدان پیوسته پاسخ بر روی بیش از یکصد هزار نقطه محاسباتی تنها بر مبنای ۲۰۰ نقطه داده اولیه و مرزی صورت پذیرفت. این نسبت حاکی از آن است که قید باقیمانده فیزیکی ($f \approx 0$) به عنوان یک قید هندسی-فیزیکی قدرتمند، فضای فرضیات شبکه را محدود کرده و از بیش‌برازش به طور کامل جلوگیری می‌کند.

\textbf{چالش لایه‌های با گرادیان تند.} نقشه خطای شکل \ref{fig:errmap} نشان می‌دهد که تمرکز عمده خطا در حوالی خط $x=0$ برای $t > 0.3$ رخ داده است. این پدیده مؤید آن است که تغییرات ناگهانی و نوک‌تیز در متغیر حالت، اصلی‌ترین چالش تقریب برای توابع فعال‌ساز هموار است. راهبرد نمونه‌برداری متمرکز در ناحیه شوک نقش موثری در مهار این خطا ایفا نمود.

\textbf{برتری رهیافت بدون شبکه بر روش‌های کلاسیک.} روش‌های گسسته‌سازی کلاسیک وابسته به تولید شبکه محاسباتی هستند و در ابعاد بالا با رشد سریع هزینه‌ها روبه‌رو می‌شوند. در نقطه مقابل، چارچوب شبکه آگاه از فیزیک بدون نیاز به تولید شبکه محاسباتی و مستقل از پیچیدگی‌های گسسته‌سازی، جواب پیوسته‌ای ارائه می‌دهد که در هر نقطه دلخواه زمانی و مکانی بدون نیاز به درون‌یابی قابل ارزیابی است.

\textbf{محدودیت‌های پژوهش حاضر.} تمرکز این پژوهش بر حل مسئله مستقیم معادله برگرز یک‌بعدی با مدل زمان‌پیوسته بوده است. گسترش به معادلات دوبعدی و سه‌بعدی و بررسی توابع فعال‌ساز تطبیقی از جمله مسائلی است که می‌تواند دقت تقریب را بیش از پیش بهبود بخشد.

\section{جمع‌بندی فصل}
\label{sec:ch4-summary}

در این فصل، ساختار الگوریتمی و پیاده‌سازی محاسباتی مدل زمان‌پیوسته شبکه آگاه از فیزیک در کتابخانه جکس برای حل مستقیم معادله غیرخطی برگرز تشریح شد. جزئیات خط لوله تبدیلات تابعی مشتق‌گیری خودکار، کامپایل در زمان اجرا و استراتژی آموزش ترکیبی دو مرحله‌ای ارائه گردید. نتایج نشان داد که مدل زمان‌پیوسته با ۲۰۰ نقطه داده اولیه-مرزی و ۱۰۰۰۰ نقطه هم‌مکانی به خطای نسبی \FwdRelErr\ در بازسازی کل میدان پاسخ دست یافته و با موفقیت پدیده تشکیل موج شوک را بازتولید می‌نماید. همچنین ارزیابی مقایسه‌ای با مقاله مرجع معتبر رئیسی و همکاران (۲۰۱۹) کارایی و قابلیت‌های این رهیافت را مورد تایید قرار داد.
