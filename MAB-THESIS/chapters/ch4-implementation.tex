\chapter{پیاده‌سازی و ارزیابی نتایج}
\label{ch:implementation}

در این فصل، جزئیات محاسباتی و نتایج تجربی حاصل از پیاده‌سازی چارچوب شبکه‌های عصبی آگاه از فیزیک در حل مستقیم معادله غیرخطی برگرز ارائه می‌گردد. ابتدا بستر نرم‌افزاری و ابزارهای مورد استفاده معرفی شده و سپس حل‌گر مرجع طیفی توسعه‌یافته برای معادله غیرخطی برگرز تشریح و با آزمون‌های مستقل اعتبارسنجی می‌شود. در ادامه، نتایج حل مستقیم معادله با مدل زمان‌پیوسته، تاریخچه همگرایی بهینه‌سازی ترکیبی و توزیع مکانی-زمانی خطاها گزارش و تحلیل می‌گردند. در بخش پایانی فصل، ارزیابی تطبیقی جامعی میان یافته‌های این پژوهش و نتایج مقاله معتبر مرجع رئیسی و همکاران (۲۰۱۹) به همراه تحلیل نقاط قوت و محدودیت‌های روش ارائه می‌شود.

\section{بستر محاسباتی و ابزارهای پیاده‌سازی}
\label{sec:env}

کلیه الگوریتم‌های محاسباتی این پژوهش با زبان برنامه‌نویسی پایتون نسخه ۳.۱۱ پیاده‌سازی شده‌اند. برای پیاده‌سازی ساختار شبکه عصبی، کامپایل توابع و انجام مشتق‌گیری خودکار، کتابخانه جکس\LTRfootnote{JAX} به کار گرفته شده است \cite{jax2018}. جکس امکان ترکیب تبدیل‌های تابعی نظیر مشتق‌گیری خودکار (\lr{grad})، برداری‌سازی خودکار (\lr{vmap}) و کامپایل در زمان اجرا (\lr{jit}) را بر بستر کامپایلر \lr{XLA} فراهم می‌سازد. انجام محاسبات عددی و ماتریسی با استفاده از کتابخانه‌های نامپای و سای‌پای، و رسم نمودارها با کتابخانه مت‌پلات‌لیب صورت پذیرفته است. الگوریتم بهینه‌سازی ترکیبی شامل پیاده‌سازی مستقیم بهینه‌ساز آدام و استفاده از رابط الگوریتم بهینه‌سازی مقید \lr{L-BFGS-B} در کتابخانه سای‌پای است.

تمامی مراحل آموزش و آزمایش‌های عددی بر روی واحد پردازش مرکزی (\lr{CPU}) و با دقت ممیز شناور ۶۴ بیتی (\lr{Float64}) اجرا شده‌اند تا امکان بازتولید دقیق نتایج بر روی تجهیزات استاندارد دانشگاهی فراهم باشد. به‌منظور تضمین تکرارپذیری، بذرهای تولید اعداد تصادفی ثابت نگاه داشته شده و کد کامل برنامه به همراه راهنمای اجرا در پیوست الف و پوشه اختصاصی \lr{code} ارائه گردیده است.

\section{حل‌گر مرجع طیفی و اعتبارسنجی آن}
\label{sec:reference}

جهت ارزیابی مستقل و دقیق عملکرد شبکه آگاه از فیزیک، یک حل‌گر عددی مرجع مستقل برای معادله برگرز با لزجت $\nu = 0.01/\pi$ توسعه داده شد. در این حل‌گر، گسسته‌سازی مکانی با روش طیفی فوریه بر روی ۱۰۲۴ مود با اعمال قاعده پالایش دو-سوم برای حذف خطای هم‌پوشانی\LTRfootnote{Aliasing Error} صورت گرفته است \cite{trefethen2000}. گام‌زنی زمانی با استفاده از روش تفاضل زمانی نمایی رانگه-کوتای مرتبه چهار (\lr{ETDRK4}) با طول گام زمانی $\Delta t = 0.001$ انجام می‌پذیرد که برای معادلات دارای جملات پخش سفت، پایداری فوق‌العاده‌ای ارائه می‌دهد \cite{kassam2005}. ماتریس پاسخ حاصل بر روی یک شبکه زمانی-مکانی با ابعاد $101 \times 1024$ ذخیره شده و به‌عنوان معیار حل دقیق در ارزیابی‌ها استفاده شده است.

صحت عملکرد حل‌گر مرجع با دو آزمون مستقل اثبات گردید:
\begin{enumerate}
\item \textbf{آزمون همگرایی عددی:} با نصف کردن گام زمانی ($\Delta t = 0.0005$)، بیشینه تغییر نسبی جواب در مرتبه $6.2\times 10^{-9}$ و با دو برابر کردن تفکیک مکانی ($N_x=2048$)، تغییر نسبی برابر $4.1\times 10^{-4}$ ثبت گردید که مؤید همگرایی مطلوب روش است.
\item \textbf{آزمون مقایسه با جواب نیمه‌تحلیلی کول-هوپف:} جواب معادله با تبدیل تحلیلی کول-هوپف \cite{hopf1950} از طریق انتگرال‌گیری عددی مستقیم با دقت بسیار بالا محاسبه شد. بیشترین اختلاف در ۱۹ نقطه نمونه‌برداری در زمان نهایی $t=1$ کمتر از $6\times 10^{-4}$ بوده و در نواحی دور از جبهه شوک به مرتبه $10^{-8}$ کاهش یافت. همچنین مطابقت با بسط‌های تحلیلی سری فوریه \cite{benton1972} نیز بررسی گردید.
\end{enumerate}

این آزمون‌ها نشان می‌دهند که دقت حل مرجع چندین مرتبه بزرگی فراتر از دقت هدف شبکه‌های عصبی بوده و مبنایی کاملا قابل‌اعتماد برای ارزیابی خطاست.

\section{حل مستقیم معادله برگرز}
\label{sec:forward}

در مسئله مستقیم، هدف محاسبه جواب $u(t,x)$ در دامنه مکانی $x \in [-1,1]$ و زمانی $t \in [0,1]$ با شرط اولیه $u(0,x) = -\sin(\pi x)$ و شرایط مرزی همگن دیریکله $u(t,-1) = u(t,1) = 0$ و ضریب لزجت $\nu = 0.01/\pi$ است. ساختار شبکه عصبی مورد استفاده، پرسپترون چندلایه با ۴ لایه پنهان، ۵۰ نورون در هر لایه و تابع فعال‌ساز $\tanh$ است که در مجموع شامل ۷۸۵۱ پارامتر قابل آموزش می‌باشد.

داده‌های آموزشی روی مرزها و شرط اولیه شامل ۲۰۰ نقطه ($N_u = 200$) شامل ۱۰۰ نقطه در $t=0$ و ۱۰۰ نقطه روی مرزهای مکانی است. تعداد نقاط هم‌مکانی درون دامنه برابر ۱۰۰۰۰ نقطه ($N_f = 10000$) انتخاب شد که نیمی از آن‌ها به‌صورت تصادفی یکنواخت و نیمی دیگر با توزیع نرمال متمرکز حول خط $x=0$ (ناحیه تشکیل جبهه شوک) نمونه‌برداری گردیدند. موقعیت نقاط آموزشی در شکل \ref{fig:fwdpoints} و مشخصات ساختاری آموزش در جدول \ref{tab:fwdconfig} خلاصه شده است.

بررسی نتایج مقایسه‌ای نکات مهم زیر را نشان می‌دهد:
\begin{enumerate}
\item \textbf{عمق در برابر پهنای شبکه:} در مقاله مرجع از شبکه‌ای عمیق‌تر (۹ لایه پنهان) با پهنای کم (۲۰ نورون) استفاده شده است، در حالی که در پژوهش حاضر از معماری با ۴ لایه پنهان و ۵۰ نورون استفاده گردید. معماری‌های عمیق‌تر قابلیت بالاتری در تقریب نواحی با گرادیان‌های بسیار تیز و ناپیوستگی‌های موضعی دارند که دلیل اصلی تفاوت در مرتبه خطای نهایی است.
\item \textbf{سقف تکرارهای بهینه‌سازی و سخت‌افزار:} در مقاله مرجع بهینه‌سازی \lr{L-BFGS} با تعداد تکرارهای بسیار بالا بر روی پردازنده‌های گرافیکی تخصصی تا رسیدن به حد نهایی دقت اجرا شده است، در حالی که در پژوهش حاضر به‌منظور امکان اجرای روان بر روی پردازنده‌های مرکزی معمولی دانشگاهی، تعداد تکرارها کنترل شده است.
\item \textbf{کیفیت بازسازی پروفیل شوک:} با وجود تفاوت عددی در خطای میانگین، همان‌طور که در شکل \ref{fig:snapshots} مشاهده می‌شود، مدل پیاده‌سازی‌شده در این پژوهش موقعیت زمانی و مکانی جبهه شوک و شیب شدید آن را در مقاطع مختلف $t=0.25, 0.50, 0.75$ با وفاداری کیفی و کمی بسیار بالایی بازتولید نموده است.
\end{enumerate}

\section{بحث و تحلیل نتایج}
\label{sec:discussion}

تحلیل نتایج عددی حاصل از پیاده‌سازی، ابعاد علمی و مهندسی متعددی را در خصوص رفتار شبکه‌های آگاه از فیزیک در حل معادله برگرز روشن می‌سازد:

\textbf{کارایی داده‌ای فوق‌العاده.} در مسئله مستقیم، تقریب میدان پیوسته پاسخ بر روی بیش از یکصد هزار نقطه محاسباتی تنها بر مبنای ۲۰۰ نقطه داده اولیه و مرزی صورت پذیرفت. این نسبت حاکی از آن است که قید باقیمانده فیزیکی ($f \approx 0$) به عنوان یک قید هندسی-فیزیکی قدرتمند، فضای فرضیات شبکه را محدود کرده و از بیش‌برازش به طور کامل جلوگیری می‌کند.

\textbf{چالش لایه‌های با گرادیان تند.} نقشه خطای شکل \ref{fig:errmap} نشان می‌دهد که تمرکز عمده خطا در حوالی خط $x=0$ برای $t > 0.3$ رخ داده است. این پدیده مؤید آن است که تغییرات ناگهانی و نوک‌تیز در متغیر حالت، اصلی‌ترین چالش تقریب برای توابع فعال‌ساز هموار است. راهبرد نمونه‌برداری متمرکز در ناحیه شوک نقش موثری در مهار این خطا ایفا نمود.

\textbf{مقایسه عملکرد با حل‌گرهای کلاسیک.} حل‌گرهای کلاسیک نظیر روش‌های طیفی برای مسائل مستقیم استاندارد با شرایط مرزی ساده، از نظر سرعت اجرا و دقت برتری دارند، اما نیازمند شبکه‌بندی دامنه هستند. در نقطه مقابل، چارچوب شبکه آگاه از فیزیک بدون نیاز به تولید شبکه محاسباتی و مستقل از پیچیدگی‌های گسسته‌سازی، جواب پیوسته‌ای ارائه می‌دهد که در هر نقطه دلخواه زمانی و مکانی بدون نیاز به درون‌یابی قابل ارزیابی است.

\textbf{محدودیت‌های پژوهش حاضر.} تمرکز این پژوهش بر حل مسئله مستقیم معادله برگرز یک‌بعدی با مدل زمان‌پیوسته بوده است. گسترش به معادلات دوبعدی و سه‌بعدی و بررسی توابع فعال‌ساز تطبیقی از جمله مسائلی است که می‌تواند دقت تقریب را بیش از پیش بهبود بخشد.

\section{جمع‌بندی فصل}
\label{sec:ch4-summary}

در این فصل، پیاده‌سازی محاسباتی مدل زمان‌پیوسته شبکه آگاه از فیزیک برای حل مستقیم معادله غیرخطی برگرز تشریح شد. حل‌گر مرجع طیفی با دقت بالا توسعه یافته و با جواب نیمه‌تحلیلی کول-هوپف اعتبارسنجی گردید. نتایج نشان داد که مدل زمان‌پیوسته با ۲۰۰ نقطه داده اولیه-مرزی و ۱۰۰۰۰ نقطه هم‌مکانی به خطای نسبی \FwdRelErr\ در بازسازی کل میدان پاسخ دست یافته و با موفقیت پدیده تشکیل موج شوک را بازتولید می‌نماید. همچنین ارزیابی مقایسه‌ای با مقاله مرجع رئیسی و همکاران (۲۰۱۹) کارایی و قابلیت‌های این رهیافت را مورد تایید قرار داد.
